cmake_minimum_required(VERSION 3.20)
# src/version.h is the single source of truth for the release version.
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/src/version.h" BS_VERSION_LINES REGEX "^#define BS_VERSION_")
foreach(part MAJOR MINOR PATCH)
  string(REGEX MATCH "#define BS_VERSION_${part} ([0-9]+)" _ "${BS_VERSION_LINES}")
  set(BS_VERSION_${part} "${CMAKE_MATCH_1}")
endforeach()
string(REGEX MATCH "#define BS_VERSION_STRING \"([0-9.]+)\"" _ "${BS_VERSION_LINES}")
set(BS_VERSION_TEXT "${CMAKE_MATCH_1}")
set(BS_VERSION "${BS_VERSION_MAJOR}.${BS_VERSION_MINOR}.${BS_VERSION_PATCH}")
if(NOT BS_VERSION_TEXT STREQUAL BS_VERSION)
  message(FATAL_ERROR "src/version.h: BS_VERSION_STRING (${BS_VERSION_TEXT}) does not match ${BS_VERSION}")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest" BS_MANIFEST)
if(NOT BS_MANIFEST MATCHES "assemblyIdentity version=\"${BS_VERSION}\\.0\"")
  message(FATAL_ERROR "src/app.manifest assemblyIdentity version must be ${BS_VERSION}.0")
endif()
project(BatteryScope VERSION ${BS_VERSION} LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
add_library(scope_core STATIC src/model.cpp src/platform.cpp src/energy_model.cpp src/energy_reader.cpp src/update.cpp src/lab.cpp src/lab_system.cpp src/lab_workload.cpp src/governor.cpp src/activity.cpp)
target_link_libraries(scope_core PUBLIC setupapi advapi32 psapi pdh user32 winhttp bcrypt ole32 shell32 uuid dxgi dwmapi cfgmgr32 ws2_32 gdi32)
add_executable(BatteryScope WIN32 src/app.cpp src/ui.cpp src/app.rc)
target_include_directories(BatteryScope PRIVATE src)
target_link_libraries(BatteryScope PRIVATE scope_core comctl32 shell32 uxtheme dwmapi gdiplus comdlg32 ole32 gdi32)
add_executable(BatteryScopeTests tests/tests.cpp)
target_link_libraries(BatteryScopeTests PRIVATE scope_core shell32)
if(MINGW)
  target_link_options(BatteryScope PRIVATE -municode -static -static-libgcc -static-libstdc++)
  target_link_options(BatteryScopeTests PRIVATE -municode -static -static-libgcc -static-libstdc++)
elseif(MSVC)
  foreach(target scope_core BatteryScope BatteryScopeTests)
    target_compile_options(${target} PRIVATE /utf-8 /W4)
  endforeach()
  target_link_options(BatteryScope PRIVATE /MANIFEST:NO)
  set_property(TARGET scope_core BatteryScope BatteryScopeTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
enable_testing()
# Runs from the source root so the translation coverage test can read src/.
add_test(NAME BatteryScopeTests COMMAND BatteryScopeTests WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME BatteryScopePanelTests COMMAND BatteryScope --ui-test "${CMAKE_CURRENT_BINARY_DIR}/panel-test.txt")
set_tests_properties(BatteryScopePanelTests PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)

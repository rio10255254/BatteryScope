cmake_minimum_required(VERSION 3.20)
project(BatteryScope VERSION 0.8.2 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
add_library(scope_core STATIC src/model.cpp src/platform.cpp src/energy_model.cpp src/energy_reader.cpp)
target_link_libraries(scope_core PUBLIC setupapi advapi32 psapi user32)
add_executable(BatteryScope WIN32 src/app.cpp src/app.rc)
target_include_directories(BatteryScope PRIVATE src)
target_link_libraries(BatteryScope PRIVATE scope_core comctl32 shell32 uxtheme dwmapi gdi32)
add_executable(BatteryScopeTests tests/tests.cpp)
target_link_libraries(BatteryScopeTests PRIVATE scope_core shell32)
if(MINGW)
  target_link_options(BatteryScope PRIVATE -municode -static -static-libgcc -static-libstdc++)
  target_link_options(BatteryScopeTests PRIVATE -municode -static -static-libgcc -static-libstdc++)
elseif(MSVC)
  foreach(target scope_core BatteryScope BatteryScopeTests)
    target_compile_options(${target} PRIVATE /utf-8 /W4)
  endforeach()
  target_link_options(BatteryScope PRIVATE /MANIFEST:NO)
  set_property(TARGET scope_core BatteryScope BatteryScopeTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
enable_testing()
add_test(NAME BatteryScopeTests COMMAND BatteryScopeTests)
add_test(NAME BatteryScopePanelTests COMMAND BatteryScope --ui-test "${CMAKE_CURRENT_BINARY_DIR}/panel-test.txt")
set_tests_properties(BatteryScopePanelTests PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
